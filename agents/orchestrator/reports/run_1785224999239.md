```typescript
import { Prisma, PrismaClient } from '@prisma/client';

const prisma = new PrismaClient();

// ---------------------------------------------------------------------------
// Typed error for not-found / ownership violations
// ---------------------------------------------------------------------------
export class ParcelNotFoundError extends Error {
  constructor(message = 'Parcel not found') {
    super(message);
    this.name = 'ParcelNotFoundError';
  }
}

export class JobNotFoundError extends Error {
  constructor(message = 'Job not found') {
    super(message);
    this.name = 'JobNotFoundError';
  }
}

// ---------------------------------------------------------------------------
// Input shapes
// ---------------------------------------------------------------------------
export interface CreateParcelInput {
  description: string;
  quantity: number;
}

export interface UpdateParcelInput {
  description?: string;
  quantity?: number;
}

// ---------------------------------------------------------------------------
// Helpers
// ---------------------------------------------------------------------------
async function assertJobExists(jobId: string): Promise<void> {
  const job = await prisma.job.findUnique({
    where: { id: jobId },
    select: { id: true },
  });
  if (!job) {
    throw new JobNotFoundError(`Job ${jobId} not found`);
  }
}

async function assertParcelBelongsToJob(
  jobId: string,
  parcelId: string,
): Promise<void> {
  const parcel = await prisma.parcel.findUnique({
    where: { id: parcelId },
    select: { jobId: true },
  });
  if (!parcel || parcel.jobId !== jobId) {
    throw new ParcelNotFoundError(
      `Parcel ${parcelId} not found on job ${jobId}`,
    );
  }
}

// ---------------------------------------------------------------------------
// Service functions
// ---------------------------------------------------------------------------
export async function listParcels(jobId: string) {
  await assertJobExists(jobId);
  return prisma.parcel.findMany({
    where: { jobId },
    orderBy: { createdAt: 'asc' },
  });
}

export async function createParcel(jobId: string, data: CreateParcelInput) {
  await assertJobExists(jobId);
  return prisma.parcel.create({
    data: {
      jobId,
      description: data.description,
      quantity: data.quantity,
    },
  });
}

export async function getParcel(jobId: string, parcelId: string) {
  await assertParcelBelongsToJob(jobId, parcelId);
  // Safe: ownership already verified above
  return prisma.parcel.findUniqueOrThrow({
    where: { id: parcelId },
  });
}

export async function updateParcel(
  jobId: string,
  parcelId: string,
  data: UpdateParcelInput,
) {
  await assertParcelBelongsToJob(jobId, parcelId);
  return prisma.parcel.update({
    where: { id: parcelId },
    data: {
      ...(data.description !== undefined && { description: data.description }),
      ...(data.quantity !== undefined && { quantity: data.quantity }),
    },
  });
}

export async function deleteParcel(jobId: string, parcelId: string) {
  await assertParcelBelongsToJob(jobId, parcelId);
  return prisma.parcel.delete({
    where: { id: parcelId },
  });
}
```

```typescript
// backend/src/parcels/parcels.controller.ts
import { Request, Response, NextFunction } from 'express';
import * as parcelsService from './parcels.service';
import { ParcelNotFoundError, JobNotFoundError } from './parcels.service';

// ---------------------------------------------------------------------------
// Validation helpers (no new library — plain guards)
// ---------------------------------------------------------------------------
const MAX_DESCRIPTION_LENGTH = 500;

function isNonEmptyString(value: unknown): value is string {
  return typeof value === 'string' && value.trim().length > 0;
}

function isPositiveInteger(value: unknown): value is number {
  return typeof value === 'number' && Number.isInteger(value) && value >= 1;
}

interface ValidationError {
  field: string;
  message: string;
}

function validateCreateBody(body: unknown): {
  errors: ValidationError[];
  description: string;
  quantity: number;
} {
  const errors: ValidationError[] = [];
  const b = body as Record<string, unknown>;

  if (!isNonEmptyString(b?.description)) {
    errors.push({
      field: 'description',
      message: 'description is required and must be a non-empty string',
    });
  } else if ((b.description as string).trim().length > MAX_DESCRIPTION_LENGTH) {
    errors.push({
      field: 'description',
      message: `description must be at most ${MAX_DESCRIPTION_LENGTH} characters`,
    });
  }

  // quantity defaults to 1 if omitted; must be positive integer if provided
  let quantity = 1;
  if (b?.quantity !== undefined) {
    if (!isPositiveInteger(b.quantity)) {
      errors.push({
        field: 'quantity',
        message: 'quantity must be a positive integer (≥ 1)',
      });
    } else {
      quantity = b.quantity as number;
    }
  }

  return {
    errors,
    description: isNonEmptyString(b?.description)
      ? (b.description as string).trim()
      : '',
    quantity,
  };
}

function validateUpdateBody(body: unknown): {
  errors: ValidationError[];
  data: parcelsService.UpdateParcelInput;
} {
  const errors: ValidationError[] = [];
  const b = body as Record<string, unknown>;
  const data: parcelsService.UpdateParcelInput = {};

  if (b?.description !== undefined) {
    if (!isNonEmptyString(b.description)) {
      errors.push({
        field: 'description',
        message: 'description must be a non-empty string',
      });
    } else if ((b.description as string).trim().length > MAX_DESCRIPTION_LENGTH) {
      errors.push({
        field: 'description',
        message: `description must be at most ${MAX_DESCRIPTION_LENGTH} characters`,
      });
    } else {
      data.description = (b.description as string).trim();
    }
  }

  if (b?.quantity !== undefined) {
    if (!isPositiveInteger(b.quantity)) {
      errors.push({
        field: 'quantity',
        message: 'quantity must be a positive integer (≥ 1)',
      });
    } else {
      data.quantity = b.quantity as number;
    }
  }

  if (Object.keys(data).length === 0 && errors.length === 0) {
    errors.push({
      field: 'body',
      message: 'at least one of description or quantity must be provided',
    });
  }

  return { errors, data };
}

// ---------------------------------------------------------------------------
// Error mapper
// ---------------------------------------------------------------------------
function handleServiceError(
  err: unknown,
  res: Response,
  next: NextFunction,
): void {
  if (err instanceof JobNotFoundError) {
    res.status(404).json({ error: err.message });
    return;
  }
  if (err instanceof ParcelNotFoundError) {
    res.status(404).json({ error: err.message });
    return;
  }
  next(err); // unhandled — delegate to Express error handler
}

// ---------------------------------------------------------------------------
// Handlers
// ---------------------------------------------------------------------------
export async function listParcels(
  req: Request,
  res: Response,
  next: NextFunction,
): Promise<void> {
  try {
    const { id: jobId } = req.params;
    const parcels = await parcelsService.listParcels(jobId);
    res.status(200).json(parcels);
  } catch (err) {
    handleServiceError(err, res, next);
  }
}

export async function createParcel(
  req: Request,
  res: Response,
  next: NextFunction,
): Promise<void> {
  try {
    const { id: jobId } = req.params;
    const { errors, description, quantity } = validateCreateBody(req.body);

    if (errors.length > 0) {
      res.status(400).json({ errors });
      return;
    }

    const parcel = await parcelsService.createParcel(jobId, {
      description,
      quantity,
    });
    res.status(201).json(parcel);
  } catch (err) {
    handleServiceError(err, res, next);
  }
}

export async function updateParcel(
  req: Request,
  res: Response,
  next: NextFunction,
): Promise<void> {
  try {
    const { id: jobId, parcelId } = req.params;
    const { errors, data } = validateUpdateBody(req.body);

    if (errors.length > 0) {
      res.status(400).json({ errors });
      return;
    }

    const parcel = await parcelsService.updateParcel(jobId, parcelId, data);
    res.status(200).json(parcel);
  } catch (err) {
    handleServiceError(err, res, next);
  }
}

export async function deleteParcel(
  req: Request,
  res: Response,
  next: NextFunction,
): Promise<void> {
  try {
    const { id: jobId, parcelId } = req.params;
    await parcelsService.deleteParcel(jobId, parcelId);
    res.status(204).send();
  } catch (err) {
    handleServiceError(err, res, next);
  }
}
```

### 2. Double Prisma query in `getParcel` — correctness defect under concurrent deletes

```typescript
export async function getParcel(jobId: string, parcelId: string) {
  await assertParcelBelongsToJob(jobId, parcelId);
  // Safe: ownership already verified above
  return prisma.parcel.findUniqueOrThrow({ where: { id: parcelId } });
}
```

**Issue:** `assertParcelBelongsToJob` already fetches `{ jobId }`. Then `getParcel` fetches the full record again. Between the two queries a concurrent delete could cause `findUniqueOrThrow` to throw a `P2025` (Prisma `NotFoundError`) that bypasses your `ParcelNotFoundError` handler and propagates as a 500. There's also no controller handler calling `getParcel` currently — but the pattern is wrong regardless.

**Fix:** Single query that fetches and checks in one round-trip:
```typescript
export async function getParcel(jobId: string, parcelId: string) {
  const parcel = await prisma.parcel.findUnique({ where: { id: parcelId } });
  if (!parcel || parcel.jobId !== jobId) {
    throw new ParcelNotFoundError(`Parcel ${parcelId} not found on job ${jobId}`);
  }
  return parcel;
}
```
Apply the same consolidation to `updateParcel` and `deleteParcel` — `assertParcelBelongsToJob` + the subsequent `update`/`delete` are also two round-trips with the same TOCTOU window.

```typescript
}
```
