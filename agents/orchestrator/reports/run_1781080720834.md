import { PrismaClient } from '@prisma/client';

export interface JobReassignmentService {
  reassignDriverJobsToTeam(driverId: string, teamId: string, prisma: PrismaClient): Promise<number>;
  unassignTeamJobs(teamId: string, prisma: PrismaClient): Promise<number>;
}

export const jobReassignmentService: JobReassignmentService = {
  /**
   * Reassign all jobs from a driver to a team
   * Sets teamId = teamId, assignedDriverId = null for all jobs where assignedDriverId = driverId
   */
  async reassignDriverJobsToTeam(driverId: string, teamId: string, prisma: PrismaClient): Promise<number> {
    const result = await prisma.job.updateMany({
      where: {
        assignedDriverId: driverId,
      },
      data: {
        teamId: teamId,
        assignedDriverId: null,
      },
    });
    
    return result.count;
  },

  /**
   * Unassign all jobs from a team and set status to DRAFT
   * Sets teamId = null, assignedDriverId = null, status = DRAFT for all jobs where teamId = teamId
   */
  async unassignTeamJobs(teamId: string, prisma: PrismaClient): Promise<number> {
    const result = await prisma.job.updateMany({
      where: {
        teamId: teamId,
      },
      data: {
        teamId: null,
        assignedDriverId: null,
        status: 'DRAFT',
      },
    });
    
    return result.count;
  },
};

import { PrismaClient } from '@prisma/client';
import { jobReassignmentService } from './jobService';

export interface TeamUpdateData {
  name?: string;
  memberIds?: string[];
}

export const teamService = {
  /**
   * Update team including member changes and job reassignment
   */
  async updateTeam(teamId: string, updateData: TeamUpdateData, prisma: PrismaClient) {
    return await prisma.$transaction(async (tx) => {
      // Get current team with members
      const existingTeam = await tx.team.findUnique({
        where: { id: teamId },
        include: { members: true },
      });

      if (!existingTeam) {
        throw new Error('Team not found');
      }

      // If memberIds is provided, handle member changes and job reassignment
      if (updateData.memberIds !== undefined) {
        const existingMemberIds = new Set(existingTeam.members.map(m => m.id));
        const newMemberIds = new Set(updateData.memberIds);
        
        // Find newly added members
        const addedMemberIds = updateData.memberIds.filter(id => !existingMemberIds.has(id));
        
        // Validate that all new member IDs exist as drivers
        if (addedMemberIds.length > 0) {
          const existingDrivers = await tx.driver.findMany({
            where: { id: { in: addedMemberIds } },
            select: { id: true },
          });
          
          const existingDriverIds = new Set(existingDrivers.map(d => d.id));
          const invalidDriverIds = addedMemberIds.filter(id => !existingDriverIds.has(id));
          
          if (invalidDriverIds.length > 0) {
            throw new Error(`Driver IDs not found: ${invalidDriverIds.join(', ')}`);
          }
        }

        // Update team members
        const updatedTeam = await tx.team.update({
          where: { id: teamId },
          data: {
            name: updateData.name,
            members: {
              set: updateData.memberIds.map(id => ({ id })),
            },
          },
          include: { members: true },
        });

        // Reassign jobs for newly added members
        for (const driverId of addedMemberIds) {
          await jobReassignmentService.reassignDriverJobsToTeam(driverId, teamId, tx);
        }

        return updatedTeam;
      } else {
        // Simple update without member changes
        return await tx.team.update({
          where: { id: teamId },
          data: { name: updateData.name },
          include: { members: true },
        });
      }
    });
  },

  /**
   * Delete team and unassign all associated jobs
   */
  async deleteTeam(teamId: string, prisma: PrismaClient) {
    return await prisma.$transaction(async (tx) => {
      // Verify team exists
      const existingTeam = await tx.team.findUnique({
        where: { id: teamId },
      });

      if (!existingTeam) {
        throw new Error('Team not found');
      }

      // Unassign all jobs from this team
      await jobReassignmentService.unassignTeamJobs(teamId, tx);

      // Delete the team
      return await tx.team.delete({
        where: { id: teamId },
      });
    });
  },

  /**
   * Get team by ID
   */
  async getTeamById(teamId: string, prisma: PrismaClient) {
    return await prisma.team.findUnique({
      where: { id: teamId },
      include: { members: true },
    });
  },
};

import { Router } from 'express';
import { z } from 'zod';
import { PrismaClient } from '@prisma/client';
import { teamService } from '../services/teamService';
import { adminOrDispatcher } from '../middleware/auth';

const router = Router();
const prisma = new PrismaClient();

// Validation schemas
const updateTeamSchema = z.object({
  name: z.string().min(1).max(100).nullish(),
  memberIds: z.array(z.string().uuid()).nullish(),
});

const teamParamsSchema = z.object({
  id: z.string().uuid(),
});

/**
 * PATCH /api/teams/:id
 * Update team including member management with job reassignment
 */
router.patch('/:id', adminOrDispatcher, async (req, res) => {
  try {
    // Validate params
    const paramsParse = teamParamsSchema.safeParse(req.params);
    if (!paramsParse.success) {
      return res.status(400).json({
        error: 'Invalid team ID',
        details: paramsParse.error.errors,
      });
    }

    // Validate body
    const bodyParse = updateTeamSchema.safeParse(req.body);
    if (!bodyParse.success) {
      return res.status(400).json({
        error: 'Invalid request body',
        details: bodyParse.error.errors,
      });
    }

    const { id } = paramsParse.data;
    const updateData = bodyParse.data;

    // Filter out nullish values
    const filteredUpdateData: { name?: string; memberIds?: string[] } = {};
    if (updateData.name !== null && updateData.name !== undefined) {
      filteredUpdateData.name = updateData.name;
    }
    if (updateData.memberIds !== null && updateData.memberIds !== undefined) {
      filteredUpdateData.memberIds = updateData.memberIds;
    }

    const updatedTeam = await teamService.updateTeam(id, filteredUpdateData, prisma);

    res.json({
      success: true,
      data: updatedTeam,
    });
  } catch (error) {
    console.error('Error updating team:', error);
    
    if (error instanceof Error) {
      if (error.message === 'Team not found') {
        return res.status(404).json({
          error: 'Team not found',
        });
      }
      if (error.message.includes('Driver IDs not found')) {
        return res.status(400).json({
          error: error.message,
        });
      }
    }

    res.status(500).json({
      error: 'Internal server error',
    });
  }
});

/**
 * DELETE /api/teams/:id
 * Delete team and unassign all associated jobs
 */
router.delete('/:id', adminOrDispatcher, async (req, res) => {
  try {
    // Validate params
    const paramsParse = teamParamsSchema.safeParse(req.params);
    if (!paramsParse.success) {
      return res.status(400).json({
        error: 'Invalid team ID',
        details: paramsParse.error.errors,
      });
    }

    const { id } = paramsParse.data;

    await teamService.deleteTeam(id, prisma);

    res.json({
      success: true,
      message: 'Team deleted successfully',
    });
  } catch (error) {
    console.error('Error deleting team:', error);
    
    if (error instanceof Error && error.message === 'Team not found') {
      return res.status(404).json({
        error: 'Team not found',
      });
    }

    res.status(500).json({
      error: 'Internal server error',
    });
  }
});

export default router;
