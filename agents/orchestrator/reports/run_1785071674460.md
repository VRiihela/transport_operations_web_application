// src/components/JobEditModal/JobEditModal.tsx
import React, { useState, useEffect } from 'react';
import axios from 'axios';
import styles from './JobEditModal.module.css';
import type { Job, ScheduleType } from '../../types/job';

// ---------------------------------------------------------------------------
// Types
// ---------------------------------------------------------------------------

interface JobEditModalProps {
  job: Job;
  onClose: () => void;
  onSuccess: () => void;
}

interface PatchJobBody {
  title?: string;
  description?: string | null;
  scheduledStart?: string | null;
  scheduledEnd?: string | null;
  scheduleType?: ScheduleType;
}

// ---------------------------------------------------------------------------
// Constants — closed set matching server-side Zod enum exactly
// ---------------------------------------------------------------------------

const SCHEDULE_TYPE_OPTIONS: { value: ScheduleType; label: string }[] = [
  { value: 'FIXED',    label: 'Kiinteä aika'  },
  { value: 'WINDOW',   label: 'Saapumisaika'  },
  { value: 'DURATION', label: 'Kesto'         },
];

// ---------------------------------------------------------------------------
// Helpers
// ---------------------------------------------------------------------------

function toIsoOrNull(value: string): string | null {
  if (!value || value.trim() === '') return null;
  const d = new Date(value);
  if (isNaN(d.getTime())) return null;
  return d.toISOString();
}

/**
 * Convert an ISO/datetime-local string to the format expected by
 * <input type="datetime-local">: "YYYY-MM-DDTHH:MM"
 */
function toDatetimeLocalValue(value: string | null | undefined): string {
  if (!value) return '';
  const d = new Date(value);
  if (isNaN(d.getTime())) return '';
  // Slice to 16 chars → "YYYY-MM-DDTHH:MM"
  return d.toISOString().slice(0, 16);
}

// ---------------------------------------------------------------------------
// Component
// ---------------------------------------------------------------------------

export function JobEditModal({ job, onClose, onSuccess }: JobEditModalProps): JSX.Element {
  // ── Form state ─────────────────────────────────────────────────────────────
  const [title, setTitle] = useState<string>(job.title ?? '');
  const [description, setDescription] = useState<string>(job.description ?? '');
  const [scheduledStart, setScheduledStart] = useState<string>(
    toDatetimeLocalValue(job.scheduledStart),
  );
  const [scheduledEnd, setScheduledEnd] = useState<string>(
    toDatetimeLocalValue(job.scheduledEnd),
  );
  const [scheduleType, setScheduleType] = useState<ScheduleType>(
    job.scheduleType ?? 'DURATION',
  );

  // ── UI state ───────────────────────────────────────────────────────────────
  const [endValidationError, setEndValidationError] = useState<string>('');
  const [submitError, setSubmitError] = useState<string>('');
  const [isSubmitting, setIsSubmitting] = useState<boolean>(false);

  // ── Re-initialise if the job prop changes (e.g. modal re-used for different jobs)
  useEffect(() => {
    setTitle(job.title ?? '');
    setDescription(job.description ?? '');
    setScheduledStart(toDatetimeLocalValue(job.scheduledStart));
    setScheduledEnd(
      job.scheduleType === 'FIXED' ? '' : toDatetimeLocalValue(job.scheduledEnd),
    );
    setScheduleType(job.scheduleType ?? 'DURATION');
    setEndValidationError('');
    setSubmitError('');
  }, [job]);

  // ── Handlers ───────────────────────────────────────────────────────────────

  function handleScheduleTypeChange(
    event: React.ChangeEvent<HTMLInputElement>,
  ): void {
    const newType = event.target.value as ScheduleType;
    setScheduleType(newType);
    setEndValidationError('');

    if (newType === 'FIXED') {
      // A fixed appointment has a single point in time; clear and disable end.
      setScheduledEnd('');
    }
  }

  function handleScheduledEndChange(
    event: React.ChangeEvent<HTMLInputElement>,
  ): void {
    setScheduledEnd(event.target.value);
    // Clear inline error as soon as the user starts correcting the field.
    if (endValidationError) setEndValidationError('');
  }

  async function handleSubmit(
    event: React.FormEvent<HTMLFormElement>,
  ): Promise<void> {
    event.preventDefault();
    setSubmitError('');

    // ── Frontend guard: WINDOW requires a scheduledEnd value ────────────────
    if (scheduleType === 'WINDOW' && scheduledEnd.trim() === '') {
      setEndValidationError(
        'Saapumisaika-tyyppi vaatii viimeisimmän saapumisajan.',
      );
      return;
    }

    // ── Build PATCH body ─────────────────────────────────────────────────────
    const body: PatchJobBody = {
      title: title.trim(),
      description: description.trim() || null,
      scheduledStart: toIsoOrNull(scheduledStart),
      // FIXED: always null. Others: convert or null.
      scheduledEnd: scheduleType === 'FIXED' ? null : toIsoOrNull(scheduledEnd),
      scheduleType,
    };

    setIsSubmitting(true);
    try {
      await axios.patch(`/api/jobs/${job.id}`, body);
      onSuccess();
      onClose();
    } catch (error: unknown) {
      if (axios.isAxiosError(error)) {
        const message =
          (error.response?.data as { message?: string } | undefined)
            ?.message ?? 'Tallennus epäonnistui.';
        setSubmitError(message);
      } else {
        setSubmitError('Odottamaton virhe. Yritä uudelleen.');
      }
    } finally {
      setIsSubmitting(false);
    }
  }

  // ── Derived values ─────────────────────────────────────────────────────────

  const scheduledEndLabel =
    scheduleType === 'WINDOW' ? 'Viimeisin saapumisaika' : 'Päättymisaika';

  const isEndDisabled = scheduleType === 'FIXED';

  // ── Render ─────────────────────────────────────────────────────────────────

  return (
    <div
      className={styles.overlay}
      role="dialog"
      aria-modal="true"
      aria-labelledby="job-edit-modal-title"
    >
      <div className={styles.modal}>
        <h2 id="job-edit-modal-title" className={styles.modalTitle}>
          Muokkaa työtä
        </h2>

        <form onSubmit={handleSubmit} noValidate>
          {/* ── Title ──────────────────────────────────────────────────────── */}
          <div className={styles.fieldGroup}>
            <label htmlFor="job-title" className={styles.label}>
              Otsikko
            </label>
            <input
              id="job-title"
              type="text"
              className={styles.input}
              value={title}
              onChange={(e) => setTitle(e.target.value)}
              required
              maxLength={255}
            />
          </div>

          {/* ── Description ────────────────────────────────────────────────── */}
          <div className={styles.fieldGroup}>
            <label htmlFor="job-description" className={styles.label}>
              Kuvaus
            </label>
            <textarea
              id="job-description"
              className={styles.textarea}
              value={description}
              onChange={(e) => setDescription(e.target.value)}
              rows={3}
              maxLength={2000}
            />
          </div>

          {/* ── Schedule type radio group ───────────────────────────────────── */}
          <fieldset className={styles.radioGroup}>
            <legend className={styles.radioGroupLegend}>Aikataulutyyppi</legend>
            {SCHEDULE_TYPE_OPTIONS.map(({ value, label }) => (
              <label key={value} className={styles.radioOption}>
                <input
                  type="radio"
                  name="scheduleType"
                  value={value}
                  checked={scheduleType === value}
                  onChange={handleScheduleTypeChange}
                />
                <span>{label}</span>
              </label>
            ))}
          </fieldset>

          {/* ── Scheduled start ─────────────────────────────────────────────── */}
          <div className={styles.fieldGroup}>
            <label htmlFor="job-scheduled-start" className={styles.label}>
              Alkamisaika
            </label>
            <input
              id="job-scheduled-start"
              type="datetime-local"
              className={styles.input}
              value={scheduledStart}
              onChange={(e) => setScheduledStart(e.target.value)}
            />
          </div>

          {/* ── Scheduled end ───────────────────────────────────────────────── */}
          <div className={styles.fieldGroup}>
            <label
              htmlFor="job-scheduled-end"
              className={styles.label}
            >
              {scheduledEndLabel}
            </label>
            <input
              id="job-scheduled-end"
              type="datetime-local"
              className={styles.input}
              value={scheduledEnd}
              onChange={handleScheduledEndChange}
              disabled={isEndDisabled}
              aria-disabled={isEndDisabled}
              aria-describedby={
                endValidationError ? 'scheduled-end-error' : undefined
              }
            />
            {endValidationError && (
              <span
                id="scheduled-end-error"
                role="alert"
                className={styles.validationError}
              >
                {endValidationError}
              </span>
            )}
          </div>

          {/* ── Submit error ────────────────────────────────────────────────── */}
          {submitError && (
            <p role="alert" className={styles.submitError}>
              {submitError}
            </p>
          )}

          {/* ── Actions ─────────────────────────────────────────────────────── */}
          <div className={styles.actions}>
            <button
              type="button"
              className={styles.cancelButton}
              onClick={onClose}
              disabled={isSubmitting}
            >
              Peruuta
            </button>
            <button
              type="submit"
              className={styles.submitButton}
              disabled={isSubmitting}
            >
              {isSubmitting ? 'Tallennetaan…' : 'Tallenna'}
            </button>
          </div>
        </form>
      </div>
    </div>
  );
}

export default JobEditModal;

function toIsoOrNull(value: string): string | null {
