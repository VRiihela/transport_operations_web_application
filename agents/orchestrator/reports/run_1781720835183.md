// src/utils/weekUtils.ts

export const FINNISH_WEEKDAY_ABBRS: readonly string[] = [
  'Ma', 'Ti', 'Ke', 'To', 'Pe', 'La', 'Su',
];

/**
 * Returns a new Date set to the Monday of the ISO week containing `date`.
 * Time is zeroed to midnight local time.
 */
export function getMonday(date: Date): Date {
  const d = new Date(date);
  d.setHours(0, 0, 0, 0);
  // getDay() returns 0=Sun,1=Mon,...,6=Sat
  const day = d.getDay();
  // ISO week: Mon=1 ... Sun=7; convert Sun(0) to 7
  const isoDay = day === 0 ? 7 : day;
  d.setDate(d.getDate() - (isoDay - 1));
  return d;
}

/**
 * Returns an array of 7 Date objects representing Mon–Sun of the week
 * starting at `monday`. `monday` must already be a Monday (use getMonday first).
 */
export function getDaysOfWeek(monday: Date): Date[] {
  return Array.from({ length: 7 }, (_, i) => {
    const d = new Date(monday);
    d.setDate(monday.getDate() + i);
    return d;
  });
}

/**
 * Formats a date as Finnish-style "D.M." with no leading zeros.
 * e.g. 2026-06-16 → "16.6."
 */
function toFinnishShortDate(date: Date): string {
  return `${date.getDate()}.${date.getMonth() + 1}.`;
}

/**
 * Returns a Finnish week range label for the week starting at `monday`.
 * Format: "D.M. – D.M.YYYY"
 * e.g. "16.6. – 22.6.2026"
 */
export function formatFinnishWeekLabel(monday: Date): string {
  const days = getDaysOfWeek(monday);
  const sunday = days[6];
  const startPart = toFinnishShortDate(monday);
  const endDay = sunday.getDate();
  const endMonth = sunday.getMonth() + 1;
  const endYear = sunday.getFullYear();
  return `${startPart} – ${endDay}.${endMonth}.${endYear}`;
}

/**
 * Returns a Finnish day-column header string.
 * e.g. abbr="Ma", date=2026-06-16 → "Ma 16.6."
 */
export function formatFinnishDayHeader(date: Date, abbr: string): string {
  return `${abbr} ${toFinnishShortDate(date)}`;
}

/**
 * Returns an ISO date string "YYYY-MM-DD" suitable for query params.
 * Uses local date components to avoid UTC-offset surprises.
 */
export function toISODateString(date: Date): string {
  const yyyy = date.getFullYear();
  const mm = String(date.getMonth() + 1).padStart(2, '0');
  const dd = String(date.getDate()).padStart(2, '0');
  return `${yyyy}-${mm}-${dd}`;
}

/**
 * Returns the Sunday (last day) of the week starting at `monday`.
 */
export function getSunday(monday: Date): Date {
  const d = new Date(monday);
  d.setDate(monday.getDate() + 6);
  return d;
}

/**
 * Adds `days` to a date, returning a new Date. `days` may be negative.
 */
export function addDays(date: Date, days: number): Date {
  const d = new Date(date);
  d.setDate(date.getDate() + days);
  return d;
}

  jobs,
  monday,
  onPrev,
  onNext,
  onDatePick,
}) => {
  const days = getDaysOfWeek(monday);

    (e: React.ChangeEvent<HTMLInputElement>) => {

      {/* Navigation row */}

The correct Sunday that maps to May is e.g. `new Date(2026, 5, 7)` (Sunday 7 Jun → Mon 1 Jun) or `new Date(2026, 5, 0)` isn't valid. A true cross-month example would be Sun 31 May 2026 → Mon 25 May 2026. Fix: use `new Date(2026, 4, 31)` (31 May 2026, a Sunday).

### 3. `getDaysOfWeek` — Fragile Consecutive-Days Test
```typescript
it('days are consecutive', () => {
  ...
  expect(days[i].getDate()).toBe(days[i - 1].getDate() + 1 <= 30
    ? days[i - 1].getDate() + 1
    : days[i].getDate()); // just check they advance
```
This conditional is incorrect logic — it allows the assertion to trivially pass at month boundaries (it just checks `days[i].getDate()` equals itself). The millisecond diff check below it (`toBe(24 * 60 * 60 * 1000)`) is the only real check, and it will **fail during DST transitions** (spring forward gives 23h, fall back gives 25h). The date check should be removed or replaced with a proper calendar check. The ms diff check should account for DST or use date component comparison only.

### 6. `WeeklyView.tsx` — No Null Guard on Date Parse
```typescript

```

This is acceptable but fragile — a `Number.isFinite` check would be more explicit and readable.
