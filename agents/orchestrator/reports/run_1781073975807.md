```typescript
import PDFDocument from 'pdfkit';
import { Job, CompletionReport } from '@prisma/client';

interface JobWithCompletionReport extends Job {
  completionReport: CompletionReport | null;
}

export class PDFService {
  static generateCompletionReportPDF(job: JobWithCompletionReport): PDFDocument {
    if (!job.completionReport || !job.completionReport.approvedAt) {
      throw new Error('No approved completion report found');
    }

    const doc = new PDFDocument({ margin: 50 });
    const report = job.completionReport;

    // Header with placeholder for company logo
    doc.fontSize(20).text('Work Management System', 50, 50);
    doc.fontSize(14).text('Completion Report', 50, 80);
    
    // Add placeholder space for future company logo
    doc.rect(400, 50, 100, 50).stroke();
    doc.fontSize(10).text('Logo Space', 415, 75);

    // Job Details Section
    let yPosition = 150;
    doc.fontSize(16).text('Job Details', 50, yPosition);
    yPosition += 30;

    const jobDetails = [
      ['Job ID:', job.id.toString()],
      ['Job Title:', job.title],
      ['Pickup Address:', job.pickupAddress],
      ['Scheduled Start:', job.scheduledStart.toLocaleString()],
      ['Scheduled End:', job.scheduledEnd.toLocaleString()]
    ];

    jobDetails.forEach(([label, value]) => {
      doc.fontSize(12).text(`${label} ${value}`, 50, yPosition);
      yPosition += 20;
    });

    // Completion Report Section
    yPosition += 20;
    doc.fontSize(16).text('Completion Report', 50, yPosition);
    yPosition += 30;

    const reportDetails = [
      ['Work Description:', report.workDescription],
      ['Customer Name:', report.customerName],
      ['Actual Start:', report.actualStart.toLocaleString()],
      ['Actual End:', report.actualEnd.toLocaleString()],
      ['Total Hours:', `${report.totalHours} hours`],
      ['Approved At:', report.approvedAt.toLocaleString()]
    ];

    reportDetails.forEach(([label, value]) => {
      doc.fontSize(12).text(`${label} ${value}`, 50, yPosition);
      yPosition += 20;
    });

    // Customer Signature Section
    if (report.customerSignature) {
      yPosition += 30;
      doc.fontSize(16).text('Customer Signature', 50, yPosition);
      yPosition += 20;

      try {
        // Decode base64 signature (expect format: data:image/[type];base64,[data])
        const base64Match = report.customerSignature.match(/^data:image\/[^;]+;base64,(.+)$/);
        if (base64Match) {
          const imageBuffer = Buffer.from(base64Match[1], 'base64');
          doc.image(imageBuffer, 50, yPosition, { width: 200, height: 100 });
        } else {
          doc.fontSize(12).text('Invalid signature format', 50, yPosition);
        }
      } catch (error) {
        doc.fontSize(12).text('Unable to display signature', 50, yPosition);
      }
    }

    doc.end();
    return doc;
  }
}
```

```typescript
// Add this import at the top
import { PDFService } from '../services/pdfService';

// Add this method to the existing jobController class/exports
export const generateCompletionReportPDF = async (req: Request, res: Response): Promise<void> => {
  try {
    // Validate job ID parameter
    const jobIdSchema = z.object({
      id: z.string().transform((val) => {
        const num = parseInt(val, 10);
        if (isNaN(num)) throw new Error('Invalid job ID');
        return num;
      })
    });

    const { id: jobId } = jobIdSchema.parse(req.params);

    // Get user from auth middleware (assumes this is set by existing auth)
    const user = req.user;
    if (!user) {
      res.status(401).json({ error: 'Unauthorized' });
      return;
    }

    // Fetch job with completion report using existing jobInclude pattern
    const job = await prisma.job.findUnique({
      where: { id: jobId },
      include: {
        ...jobInclude,
        completionReport: true,
        assignedDriver: true,
        team: {
          include: {
            members: true
          }
        }
      }
    });

    if (!job) {
      res.status(404).json({ error: 'Job not found' });
      return;
    }

    // Check if completion report exists and is approved
    if (!job.completionReport || !job.completionReport.approvedAt) {
      res.status(404).json({ error: 'No approved completion report found' });
      return;
    }

    // Role-based access control
    const userRole = user.role;
    const userId = user.id;

    if (userRole === 'DRIVER') {
      // Driver can only access jobs assigned to them or their team
      const isAssignedDriver = job.assignedDriverId === userId;
      const isTeamMember = job.team?.members.some(member => member.id === userId);

      if (!isAssignedDriver && !isTeamMember) {
        res.status(403).json({ error: 'Access denied: Job not assigned to you or your team' });
        return;
      }
    }
    // Admin and Dispatcher have unrestricted access - no additional checks needed

    // Generate PDF
    const pdfDoc = PDFService.generateCompletionReportPDF(job);

    // Set response headers for PDF download
    res.setHeader('Content-Type', 'application/pdf');
    res.setHeader('Content-Disposition', `attachment; filename=completion-report-${jobId}.pdf`);

    // Stream PDF to response
    pdfDoc.pipe(res);

  } catch (error) {
    if (error instanceof z.ZodError) {
      res.status(400).json({ error: 'Invalid job ID format' });
      return;
    }

    console.error('PDF generation error:', error);
    res.status(500).json({ error: 'Failed to generate PDF' });
  }
};
```
