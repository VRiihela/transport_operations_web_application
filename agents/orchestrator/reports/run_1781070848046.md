```typescript
import React, { useState, useEffect } from 'react';
import { useAuth } from '../../contexts/AuthContext';
import { axiosInstance } from '../../services/api';
import { Job, CompletionReport } from '../../types/job';
import styles from './CompletionModal.module.css';

interface CompletionModalProps {
  job: Job;
  isOpen: boolean;
  onClose: () => void;
  onSubmit: (data: CompletionReport) => Promise<void>;
  onUpdate?: (updatedJob: Job) => void;
}

interface UnlockResponse {
  success: boolean;
  completionReport: CompletionReport;
  message?: string;
}

interface UnlockError {
  message: string;
  code?: string;
}

export const CompletionModal: React.FC<CompletionModalProps> = ({
  job,
  isOpen,
  onClose,
  onSubmit,
  onUpdate
}) => {
  const { user } = useAuth();
  const [isUnlocking, setIsUnlocking] = useState<boolean>(false);
  const [unlockError, setUnlockError] = useState<string | null>(null);
  const [formData, setFormData] = useState<CompletionReport>({
    notes: job.completionReport?.notes || '',
    completedAt: job.completionReport?.completedAt || new Date().toISOString(),
    approvedAt: job.completionReport?.approvedAt || null
  });

  // Reset unlock error when modal opens/closes
  useEffect(() => {
    if (isOpen) {
      setUnlockError(null);
    }
  }, [isOpen]);

  const isLocked = job.completionReport?.approvedAt !== null && job.completionReport?.approvedAt !== undefined;
  const isAdmin = user?.role === 'admin';

  const handleUnlock = async (): Promise<void> => {
    if (!job.id || !isAdmin) return;

    setIsUnlocking(true);
    setUnlockError(null);

    try {
      const response = await axiosInstance.post<UnlockResponse>(
        `/api/jobs/${job.id}/completion-report/unlock`
      );

      if (response.data.success) {
        // Update the job state with the unlocked completion report
        const updatedJob: Job = {
          ...job,
          completionReport: response.data.completionReport
        };

        // Update form data to reflect unlocked state
        setFormData({
          notes: response.data.completionReport.notes || '',
          completedAt: response.data.completionReport.completedAt || new Date().toISOString(),
          approvedAt: response.data.completionReport.approvedAt || null
        });

        // Notify parent component of the update
        if (onUpdate) {
          onUpdate(updatedJob);
        }
      } else {
        setUnlockError(response.data.message || 'Failed to unlock completion report');
      }
    } catch (error: any) {
      let errorMessage = 'An error occurred while unlocking';
      
      if (error.response?.data?.message) {
        errorMessage = error.response.data.message;
      } else if (error.message) {
        errorMessage = error.message;
      }
      
      setUnlockError(errorMessage);
    } finally {
      setIsUnlocking(false);
    }
  };

  const handleSubmit = async (e: React.FormEvent): Promise<void> => {
    e.preventDefault();
    
    if (isLocked) {
      return; // Prevent submission when locked
    }

    try {
      await onSubmit(formData);
    } catch (error) {
      // Error handling is managed by parent component
      console.error('Submission error:', error);
    }
  };

  const handleInputChange = (e: React.ChangeEvent<HTMLTextAreaElement>): void => {
    const { name, value } = e.target;
    setFormData(prev => ({
      ...prev,
      [name]: value
    }));
  };

  if (!isOpen) return null;

  return (
    <div className={styles.overlay}>
      <div className={styles.modal}>
        <div className={styles.header}>
          <h2>Completion Report</h2>
          <button
            type="button"
            className={styles.closeButton}
            onClick={onClose}
            aria-label="Close modal"
          >
            ×
          </button>
        </div>

        {/* Locked Banner */}
        {isLocked && (
          <div className={styles.lockedBanner}>
            <div className={styles.lockedMessage}>
              <span className={styles.lockIcon}>🔒</span>
              <span>Approved — locked for editing</span>
            </div>
            
            {isAdmin && (
              <div className={styles.unlockSection}>
                {unlockError && (
                  <div className={styles.unlockError} role="alert">
                    {unlockError}
                  </div>
                )}
                <button
                  type="button"
                  className={styles.unlockButton}
                  onClick={handleUnlock}
                  disabled={isUnlocking}
                  aria-label="Unlock completion report for editing"
                >
                  {isUnlocking ? 'Unlocking...' : 'Unlock'}
                </button>
              </div>
            )}
          </div>
        )}

        <form onSubmit={handleSubmit} className={styles.form}>
          <div className={styles.field}>
            <label htmlFor="notes" className={styles.label}>
              Completion Notes
            </label>
            <textarea
              id="notes"
              name="notes"
              value={formData.notes}
              onChange={handleInputChange}
              className={styles.textarea}
              rows={6}
              disabled={isLocked}
              placeholder="Enter completion details..."
              required
            />
          </div>

          <div className={styles.actions}>
            <button
              type="button"
              className={styles.cancelButton}
              onClick={onClose}
            >
              Cancel
            </button>
            <button
              type="submit"
              className={styles.submitButton}
              disabled={isLocked}
            >
              Submit Report
            </button>
          </div>
        </form>
      </div>
    </div>
  );
};
```
