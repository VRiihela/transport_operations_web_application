import React from 'react';
import { Job } from '../types';
import styles from './JobCard.module.css';

interface JobCardProps {
  job: Job;
  onClick: (job: Job) => void;
}

const JobCard: React.FC<JobCardProps> = ({ job, onClick }) => {
  const handleClick = () => {
    onClick(job);
  };

  const renderPickupAddress = () => {
    const { pickupAddress, customerName } = job;
    
    if (!pickupAddress?.street && !pickupAddress?.postalCode && !pickupAddress?.city && !customerName) {
      return null;
    }

    const addressLines: string[] = [];
    
    // First line: street
    if (pickupAddress?.street) {
      addressLines.push(pickupAddress.street);
    }
    
    // Second line: postal code + city + customer name
    const secondLineItems: string[] = [];
    if (pickupAddress?.postalCode) {
      secondLineItems.push(pickupAddress.postalCode);
    }
    if (pickupAddress?.city) {
      secondLineItems.push(pickupAddress.city);
    }
    if (customerName) {
      secondLineItems.push(`(${customerName})`);
    }
    
    if (secondLineItems.length > 0) {
      addressLines.push(secondLineItems.join(' '));
    }

    if (addressLines.length === 0) {
      return null;
    }

    return (
      <div className={styles.addressContainer}>
        {addressLines.map((line, index) => (
          <div key={index} className={styles.addressLine}>
            {line}
          </div>
        ))}
      </div>
    );
  };

  const getPriorityClass = (priority: string) => {
    switch (priority) {
      case 'high':
        return styles.priorityHigh;
      case 'medium':
        return styles.priorityMedium;
      case 'low':
        return styles.priorityLow;
      default:
        return styles.priorityMedium;
    }
  };

  const getStatusClass = (status: string) => {
    switch (status) {
      case 'pending':
        return styles.statusPending;
      case 'in_progress':
        return styles.statusInProgress;
      case 'completed':
        return styles.statusCompleted;
      default:
        return styles.statusPending;
    }
  };

  return (
    <div className={styles.jobCard} onClick={handleClick}>
      <div className={styles.titleRow}>
        <h3 className={styles.title}>{job.title}</h3>
        <span className={`${styles.priority} ${getPriorityClass(job.priority)}`}>
          {job.priority}
        </span>
      </div>
      
      {renderPickupAddress()}
      
      <p className={styles.description}>{job.description}</p>
      
      <div className={styles.metadata}>
        <span className={`${styles.status} ${getStatusClass(job.status)}`}>
          {job.status.replace('_', ' ')}
        </span>
        {job.assignedTo && <span>Assigned: {job.assignedTo}</span>}
        {job.dueDate && <span>Due: {new Date(job.dueDate).toLocaleDateString()}</span>}
      </div>
    </div>
  );
};

export default JobCard;
