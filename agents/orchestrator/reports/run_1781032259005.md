export const updateCompletionReport = async (req: Request, res: Response): Promise<void> => {
  try {
    const jobId = parseInt(req.params.id);
    const userId = req.user!.id;

    // Validate job ID
    if (isNaN(jobId)) {
      res.status(400).json({ error: 'Invalid job ID' });
      return;
    }

    // Check if job exists and get completion report
    const job = await prisma.job.findUnique({
      where: { id: jobId },
      select: {
        id: true,
        completionReport: {
          select: {
            id: true,
            approvedAt: true
          }
        }
      }
    });

    if (!job) {
      res.status(404).json({ error: 'Job not found' });
      return;
    }

    // Check if completion report is locked
    if (job.completionReport?.approvedAt) {
      res.status(403).json({ 
        error: 'Completion report is locked after approval. Admin must unlock before editing.' 
      });
      return;
    }

    // Validate request body with Zod
    const validatedData = updateCompletionReportSchema.parse(req.body);

    // Update completion report
    const updatedJob = await prisma.job.update({
      where: { id: jobId },
      data: {
        completionReport: {
          upsert: {
            create: {
              ...validatedData,
              createdBy: userId
            },
            update: validatedData
          }
        }
      },
      select: jobInclude.select
    });

    // Audit log the update
    await auditService.log({
      event: AuditEvent.COMPLETION_REPORT_UPDATED,
      userId,
      details: { jobId }
    });

    res.json({ data: updatedJob });
  } catch (error) {
    if (error instanceof z.ZodError) {
      res.status(400).json({ error: 'Invalid request data', details: error.errors });
      return;
    }
    console.error('Error updating completion report:', error);
    res.status(500).json({ error: 'Internal server error' });
  }
};

export const approveCompletionReport = async (req: Request, res: Response): Promise<void> => {
  try {
    const jobId = parseInt(req.params.id);
    const userId = req.user!.id;

    // Validate job ID
    if (isNaN(jobId)) {
      res.status(400).json({ error: 'Invalid job ID' });
      return;
    }

    // Check if job exists and get completion report
    const job = await prisma.job.findUnique({
      where: { id: jobId },
      select: {
        id: true,
        completionReport: {
          select: {
            id: true,
            approvedAt: true
          }
        }
      }
    });

    if (!job) {
      res.status(404).json({ error: 'Job not found' });
      return;
    }

    if (!job.completionReport) {
      res.status(404).json({ error: 'Completion report not found' });
      return;
    }

    // Check if already approved
    if (job.completionReport.approvedAt) {
      res.status(403).json({ 
        error: 'Completion report is already approved' 
      });
      return;
    }

    // Approve the completion report
    const updatedJob = await prisma.job.update({
      where: { id: jobId },
      data: {
        completionReport: {
          update: {
            approvedAt: new Date(),
            approvedBy: userId
          }
        }
      },
      select: jobInclude.select
    });

    // Audit log the approval
    await auditService.log({
      event: AuditEvent.COMPLETION_REPORT_APPROVED,
      userId,
      details: { jobId }
    });

    res.json({ data: updatedJob });
  } catch (error) {
    console.error('Error approving completion report:', error);
    res.status(500).json({ error: 'Internal server error' });
  }
};
