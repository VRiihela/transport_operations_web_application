// ── was looking at.  Initialize to the Monday of the current week.
function getCurrentWeekMonday(): Date {
  const now = new Date();
  const day = now.getDay(); // 0 = Sun
  const diff = day === 0 ? -6 : 1 - day; // shift to Monday
  const monday = new Date(now);
  monday.setDate(now.getDate() + diff);
  monday.setHours(0, 0, 0, 0);
  return monday;
}

export function DispatcherBoard(): JSX.Element {

  // ── Lifted from ScheduleView so it survives view switches

    getCurrentWeekMonday,

  // ── click can pre-set it before switching views.

## 2. Correctness Issues

### 2.1 `makeKeyDownHandler` recreates functions on every render despite `useCallback`
```tsx
const makeKeyDownHandler = useCallback(
  (date: Date) => (event: KeyboardEvent<HTMLDivElement>): void => { ... },
  [onDayHeaderClick],
);
```
`makeKeyDownHandler` is stable, but it is called inline in JSX — `onKeyDown={makeKeyDownHandler(col.date)}` — which creates a new function on each render for each column. This is not a correctness bug but it negates the `useCallback`. Minor but worth noting for 7 columns × every render.

**Preferred pattern:**
```tsx
onKeyDown={(e) => { if (e.key === 'Enter' || e.key === ' ') { e.preventDefault(); onDayHeaderClick(col.date); } }}
```
Or extract a `DayHeader` sub-component so each column's handler is stable.

### 2.2 `role="button"` on a non-`<button>` element without `<button>`
The architect calls for `role="button"` + `tabIndex={0}`. A native `<button>` is preferred over a `<div role="button">` because it provides keyboard activation (Enter) natively, focus management, and no need for manual `onKeyDown`. ### 2.3 `getCurrentWeekMonday` does not handle Sunday correctly — edge case
```tsx
const diff = day === 0 ? -6 : 1 - day;
```
If today is Sunday (`day === 0`), this gives Monday 6 days ago. That is correct for ISO weeks (Monday-start). However the `buildWeekColumns` helper starts from `weekStart` and iterates 0–6, which for a Monday start gives Mon–Sun. If the existing `ScheduleView` uses Sunday-start weeks, this is a behavioral regression. **Must be validated against real implementation.**

### 2.4 `date.toISOString()` used as React key — timezone hazard
```tsx
key={col.date.toISOString()}
```
`toISOString()` returns UTC. For dates created with `setDate` in local time, this can shift the date string by a day for users in UTC+N timezones (e.g. `2024-10-07T00:00:00.000Z` becomes `2024-10-06T...` in UTC+3). Using `col.date.toLocaleDateString()` or a `YYYY-MM-DD` formatter is safer for keys and for `data-testid`.
