import { axiosInstance } from './api';
import { Job } from '../types/job';

// ... existing code ...

export const downloadCompletionReportPdf = async (jobId: string): Promise<void> => {
  try {
    const response = await axiosInstance.get(`/api/jobs/${jobId}/completion-report/pdf`, {
      responseType: 'blob',
    });

    // Create blob URL
    const blob = new Blob([response.data], { type: 'application/pdf' });
    const url = URL.createObjectURL(blob);

    // Create temporary download link
    const link = document.createElement('a');
    link.href = url;
    link.download = `completion-report-${jobId}.pdf`;
    document.body.appendChild(link);
    
    // Trigger download
    link.click();
    
    // Cleanup
    document.body.removeChild(link);
    URL.revokeObjectURL(url);
  } catch (error) {
    console.error('Failed to download PDF:', error);
    throw error;
  }
};

import React, { useState } from 'react';
import { Job } from '../../types/job';
import { downloadCompletionReportPdf } from '../../services/jobService';
import styles from './CompletionModal.module.css';

interface CompletionModalProps {
  job: Job;
  onClose: () => void;
}

const CompletionModal: React.FC<CompletionModalProps> = ({ job, onClose }) => {
  const [isDownloading, setIsDownloading] = useState(false);
  const [downloadError, setDownloadError] = useState<string | null>(null);

  const handleDownloadPdf = async (): Promise<void> => {
    setIsDownloading(true);
    setDownloadError(null);
    
    try {
      await downloadCompletionReportPdf(job.id);
    } catch (error) {
      setDownloadError('Failed to download PDF. Please try again.');
    } finally {
      setIsDownloading(false);
    }
  };

  const isReportApproved = job.completionReport?.approvedAt !== null && job.completionReport?.approvedAt !== undefined;

  return (
    <div className={styles.modalOverlay} onClick={onClose}>
      <div className={styles.modalContent} onClick={(e) => e.stopPropagation()}>
        <div className={styles.modalHeader}>
          <h2>Completion Report</h2>
          <button className={styles.closeButton} onClick={onClose}>
            ×
          </button>
        </div>
        
        <div className={styles.modalBody}>
          {/* Existing completion report content */}
          {job.completionReport ? (
            <div className={styles.reportContent}>
              <div className={styles.reportField}>
                <label>Completion Notes:</label>
                <p>{job.completionReport.notes}</p>
              </div>
              
              {job.completionReport.photos && job.completionReport.photos.length > 0 && (
                <div className={styles.reportField}>
                  <label>Photos:</label>
                  <div className={styles.photoGrid}>
                    {job.completionReport.photos.map((photo, index) => (
                      <img key={index} src={photo.url} alt={`Completion ${index + 1}`} className={styles.photo} />
                    ))}
                  </div>
                </div>
              )}
              
              <div className={styles.reportField}>
                <label>Completed At:</label>
                <p>{new Date(job.completionReport.completedAt).toLocaleString()}</p>
              </div>
              
              {job.completionReport.approvedAt && (
                <div className={styles.reportField}>
                  <label>Approved At:</label>
                  <p>{new Date(job.completionReport.approvedAt).toLocaleString()}</p>
                </div>
              )}
            </div>
          ) : (
            <p>No completion report available.</p>
          )}
        </div>
        
        <div className={styles.modalFooter}>
          {isReportApproved && (
            <div className={styles.downloadSection}>
              <button
                className={styles.downloadButton}
                onClick={handleDownloadPdf}
                disabled={isDownloading}
              >
                {isDownloading ? 'Downloading...' : 'Download PDF'}
              </button>
              {downloadError && (
                <p className={styles.errorMessage}>{downloadError}</p>
              )}
            </div>
          )}
          <button className={styles.closeFooterButton} onClick={onClose}>
            Close
          </button>
        </div>
      </div>
    </div>
  );
};

export default CompletionModal;
