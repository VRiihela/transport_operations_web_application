### File: `src/components/JobCreateModal.tsx`
```typescript
import React, { useState, useEffect } from 'react';
import { createJob } from '../api/jobs';
import styles from './JobCreateModal.module.css';

interface JobCreateModalProps {
  isOpen: boolean;
  onClose: () => void;
  onJobCreated: () => void;
}

interface FormData {
  name: string;
  customerId: string;
  jobType: string;
  scheduledStart: string;
  scheduledEnd: string;
  services: string[];
  addresses: Array<{
    street: string;
    city: string;
    state: string;
    zipCode: string;
  }>;
  jobContent: string;
  dispatcherNotes: string;
}

interface FormErrors {
  general?: string;
  nameOrCustomer?: string;
  jobType?: string;
  addresses?: string;
}

const JobCreateModal: React.FC<JobCreateModalProps> = ({
  isOpen,
  onClose,
  onJobCreated,
}) => {
  const [formData, setFormData] = useState<FormData>({
    name: '',
    customerId: '',
    jobType: '',
    scheduledStart: '',
    scheduledEnd: '',
    services: [],
    addresses: [{ street: '', city: '', state: '', zipCode: '' }],
    jobContent: '',
    dispatcherNotes: '',
  });

  const [errors, setErrors] = useState<FormErrors>({});
  const [isLoading, setIsLoading] = useState(false);
  const [apiError, setApiError] = useState<string>('');

  useEffect(() => {
    if (!isOpen) {
      // Reset form when modal closes
      setFormData({
        name: '',
        customerId: '',
        jobType: '',
        scheduledStart: '',
        scheduledEnd: '',
        services: [],
        addresses: [{ street: '', city: '', state: '', zipCode: '' }],
        jobContent: '',
        dispatcherNotes: '',
      });
      setErrors({});
      setApiError('');
      setIsLoading(false);
    }
  }, [isOpen]);

  const validateForm = (): boolean => {
    const newErrors: FormErrors = {};

    // Name OR customerId required
    if (!formData.name.trim() && !formData.customerId.trim()) {
      newErrors.nameOrCustomer = 'Either job name or customer ID is required';
    }

    // Job type required
    if (!formData.jobType.trim()) {
      newErrors.jobType = 'Job type is required';
    }

    // At least one address with street + city
    const hasValidAddress = formData.addresses.some(
      (addr) => addr.street.trim() && addr.city.trim()
    );
    if (!hasValidAddress) {
      newErrors.addresses = 'At least one address must have street and city filled';
    }

    setErrors(newErrors);
    return Object.keys(newErrors).length === 0;
  };

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    setApiError('');

    if (!validateForm()) {
      return;
    }

    setIsLoading(true);

    try {
      const payload = {
        name: formData.name.trim() || null,
        customer_id: formData.customerId.trim() || null,
        job_type: formData.jobType.trim(),
        scheduled_start: formData.scheduledStart || null,
        scheduled_end: formData.scheduledEnd || null,
        services: formData.services,
        addresses: formData.addresses.filter(
          (addr) => addr.street.trim() && addr.city.trim()
        ),
        job_content: formData.jobContent.trim(),
        dispatcher_notes: formData.dispatcherNotes.trim(),
      };

      await createJob(payload);
      onJobCreated();
      onClose();
    } catch (error: any) {
      const errorMessage = error.response?.data?.error || 'Failed to create job';
      setApiError(errorMessage);
    } finally {
      setIsLoading(false);
    }
  };

  const updateFormData = (field: keyof FormData, value: any) => {
    setFormData(prev => ({ ...prev, [field]: value }));
  };

  const addAddress = () => {
    setFormData(prev => ({
      ...prev,
      addresses: [...prev.addresses, { street: '', city: '', state: '', zipCode: '' }]
    }));
  };

  const updateAddress = (index: number, field: string, value: string) => {
    setFormData(prev => ({
      ...prev,
      addresses: prev.addresses.map((addr, i) => 
        i === index ? { ...addr, [field]: value } : addr
      )
    }));
  };

  const removeAddress = (index: number) => {
    if (formData.addresses.length > 1) {
      setFormData(prev => ({
        ...prev,
        addresses: prev.addresses.filter((_, i) => i !== index)
      }));
    }
  };

  if (!isOpen) return null;

  return (
    <div className={styles.modalOverlay}>
      <div className={styles.modalContent}>
        <div className={styles.modalHeader}>
          <h2>Create New Job</h2>
          <button 
            type="button" 
            onClick={onClose}
            className={styles.closeButton}
            disabled={isLoading}
          >
            ×
          </button>
        </div>

        <form onSubmit={handleSubmit} className={styles.form}>
          {/* Basic Info Section */}
          <div className={styles.section}>
            <h3>Basic Information</h3>
            
            <div className={styles.fieldGroup}>
              <div className={styles.field}>
                <label htmlFor="jobName">Job Name</label>
                <input
                  id="jobName"
                  type="text"
                  value={formData.name}
                  onChange={(e) => updateFormData('name', e.target.value)}
                  className={errors.nameOrCustomer ? styles.errorInput : ''}
                  disabled={isLoading}
                />
              </div>
              
              <div className={styles.field}>
                <label htmlFor="customerId">Customer ID</label>
                <input
                  id="customerId"
                  type="text"
                  value={formData.customerId}
                  onChange={(e) => updateFormData('customerId', e.target.value)}
                  className={errors.nameOrCustomer ? styles.errorInput : ''}
                  disabled={isLoading}
                />
              </div>
            </div>
            
            {errors.nameOrCustomer && (
              <div className={styles.errorMessage}>{errors.nameOrCustomer}</div>
            )}

            <div className={styles.field}>
              <label htmlFor="jobType">Job Type *</label>
              <select
                id="jobType"
                value={formData.jobType}
                onChange={(e) => updateFormData('jobType', e.target.value)}
                className={errors.jobType ? styles.errorInput : ''}
                disabled={isLoading}
              >
                <option value="">Select job type...</option>
                <option value="installation">Installation</option>
                <option value="maintenance">Maintenance</option>
                <option value="repair">Repair</option>
                <option value="inspection">Inspection</option>
              </select>
              {errors.jobType && (
                <div className={styles.errorMessage}>{errors.jobType}</div>
              )}
            </div>
          </div>

          {/* Scheduling Section */}
          <div className={styles.section}>
            <h3>Schedule</h3>
            <div className={styles.fieldGroup}>
              <div className={styles.field}>
                <label htmlFor="scheduledStart">Start Date</label>
                <input
                  id="scheduledStart"
                  type="datetime-local"
                  value={formData.scheduledStart}
                  onChange={(e) => updateFormData('scheduledStart', e.target.value)}
                  disabled={isLoading}
                />
              </div>
              
              <div className={styles.field}>
                <label htmlFor="scheduledEnd">End Date</label>
                <input
                  id="scheduledEnd"
                  type="datetime-local"
                  value={formData.scheduledEnd}
                  onChange={(e) => updateFormData('scheduledEnd', e.target.value)}
                  disabled={isLoading}
                />
              </div>
            </div>
          </div>

          {/* Addresses Section */}
          <div className={styles.section}>
            <h3>Addresses</h3>
            {formData.addresses.map((address, index) => (
              <div key={index} className={styles.addressBlock}>
                <div className={styles.addressHeader}>
                  <h4>Address {index + 1}</h4>
                  {formData.addresses.length > 1 && (
                    <button
                      type="button"
                      onClick={() => removeAddress(index)}
                      className={styles.removeButton}
                      disabled={isLoading}
                    >
                      Remove
                    </button>
                  )}
                </div>
                
                <div className={styles.fieldGroup}>
                  <div className={styles.field}>
                    <label>Street</label>
                    <input
                      type="text"
                      value={address.street}
                      onChange={(e) => updateAddress(index, 'street', e.target.value)}
                      disabled={isLoading}
                    />
                  </div>
                  
                  <div className={styles.field}>
                    <label>City</label>
                    <input
                      type="text"
                      value={address.city}
                      onChange={(e) => updateAddress(index, 'city', e.target.value)}
                      disabled={isLoading}
                    />
                  </div>
                </div>
                
                <div className={styles.fieldGroup}>
                  <div className={styles.field}>
                    <label>State</label>
                    <input
                      type="text"
                      value={address.state}
                      onChange={(e) => updateAddress(index, 'state', e.target.value)}
                      disabled={isLoading}
                    />
                  </div>
                  
                  <div className={styles.field}>
                    <label>Zip Code</label>
                    <input
                      type="text"
                      value={address.zipCode}
                      onChange={(e) => updateAddress(index, 'zipCode', e.target.value)}
                      disabled={isLoading}
                    />
                  </div>
                </div>
              </div>
            ))}
            
            {errors.addresses && (
              <div className={styles.errorMessage}>{errors.addresses}</div>
            )}
            
            <button
              type="button"
              onClick={addAddress}
              className={styles.addButton}
              disabled={isLoading}
            >
              Add Another Address
            </button>
          </div>

          {/* Notes Section */}
          <div className={styles.section}>
            <h3>Notes</h3>
            
            <div className={styles.field}>
              <label htmlFor="jobContent">Job Content</label>
              <textarea
                id="jobContent"
                value={formData.jobContent}
                onChange={(e) => updateFormData('jobContent', e.target.value)}
                placeholder="What is being delivered or installed..."
                rows={3}
                disabled={isLoading}
                className={styles.textarea}
              />
            </div>
            
            <div className={styles.field}>
              <label htmlFor="dispatcherNotes">Dispatcher Notes</label>
              <textarea
                id="dispatcherNotes"
                value={formData.dispatcherNotes}
                onChange={(e) => updateFormData('dispatcherNotes', e.target.value)}
                placeholder="Internal notes for dispatchers..."
                rows={3}
                disabled={isLoading}
                className={styles.textarea}
              />
            </div>
          </div>

          {/* API Error Display */}
          {apiError && (
            <div className={styles.apiError}>
              {apiError}
            </div>
          )}

          {/* Submit Button */}
          <div className={styles.modalFooter}>
            <button
              type="button"
              onClick={onClose}
              className={styles.cancelButton}
              disabled={isLoading}
            >
              Cancel
            </button>
            
            <button
              type="submit"
              className={styles.submitButton}
              disabled={isLoading}
            >
              {isLoading ? 'Creating...' : 'Create Job'}
            </button>
          </div>
        </form>
      </div>
    </div>
  );
};

export default JobCreateModal;
```
