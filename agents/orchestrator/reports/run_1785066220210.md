// ---------------------------------------------------------------------------
// ScheduleType enum — matches Prisma enum values exactly.
// Using z.enum with string literals to avoid tight coupling to the
// Prisma-generated runtime enum object.
// ---------------------------------------------------------------------------
export const ScheduleTypeSchema = z.enum(['FIXED', 'WINDOW', 'DURATION']);
export type ScheduleType = z.infer<typeof ScheduleTypeSchema>;

// ---------------------------------------------------------------------------
// Cross-field refinement helper
// Reused in both create and update schemas.
// Rule: scheduleType WINDOW requires scheduledEnd to be a non-null value.
// ---------------------------------------------------------------------------
function requireScheduledEndForWindow<
  T extends { scheduleType?: string | null; scheduledEnd?: string | Date | null }
>(data: T, ctx: z.RefinementCtx): void {
  if (data.scheduleType === 'WINDOW' && (data.scheduledEnd === null || data.scheduledEnd === undefined)) {
    ctx.addIssue({
      code: z.ZodIssueCode.custom,
      message: 'scheduleType WINDOW requires scheduledEnd to be provided',
      path: ['scheduledEnd'],
    });
  }
}

// ---------------------------------------------------------------------------
// Create Job Schema
// ---------------------------------------------------------------------------
export const createJobSchema = z
  .object({
    // --- existing fields (preserve exactly as they were) ---
    title: z.string().min(1, 'Title is required').max(255),
    description: z.string().max(2000).nullish(),
    status: z.enum(['PENDING', 'IN_PROGRESS', 'COMPLETED', 'CANCELLED']).optional(),
    scheduledStart: z.string().datetime({ message: 'scheduledStart must be a valid ISO datetime' }).nullish(),
    scheduledEnd: z.string().datetime({ message: 'scheduledEnd must be a valid ISO datetime' }).nullish(),
    // ... other existing fields preserved verbatim ...

    // --- new field ---
    scheduleType: ScheduleTypeSchema.optional(),
    // Optional on create: omitting it lets Prisma apply @default(DURATION)

    status: z.enum(['PENDING', 'IN_PROGRESS', 'COMPLETED', 'CANCELLED']).optional(),
