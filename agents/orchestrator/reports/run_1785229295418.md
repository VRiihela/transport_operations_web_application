```typescript
import React from 'react';
import type { Parcel } from '../../../types/Job';
import t from '../translations';

interface ParcelRow {
  id: string;
  description: string;
  quantity: number;
}

interface Props {
  parcels: ParcelRow[];
  onChange: (parcels: ParcelRow[]) => void;
}

function createEmptyParcel(): ParcelRow {
  return {
    id: crypto.randomUUID(),
    description: '',
    quantity: 1,
  };
}

const ParcelsSection: React.FC<Props> = ({ parcels, onChange }) => {
  const handleAdd = () => {
    onChange([...parcels, createEmptyParcel()]);
  };

  const handleRemove = (id: string) => {
    onChange(parcels.filter((p) => p.id !== id));
  };

  const handleDescriptionChange = (id: string, value: string) => {
    onChange(
      parcels.map((p) =>
        p.id === id ? { ...p, description: value } : p
      )
    );
  };

  const handleQuantityChange = (id: string, raw: string) => {
    const parsed = parseInt(raw, 10);
    const quantity = Number.isNaN(parsed) || parsed < 1 ? 1 : parsed;
    onChange(
      parcels.map((p) =>
        p.id === id ? { ...p, quantity } : p
      )
    );
  };

  return (
    <fieldset>
      <legend>{t.fieldParcels}</legend>

      {parcels.map((parcel, index) => (
        <div key={parcel.id} role="group" aria-label={`${t.fieldParcels} ${index + 1}`}>
          <label htmlFor={`parcel-description-${parcel.id}`}>
            {t.fieldParcelDescription}
          </label>
          <input
            id={`parcel-description-${parcel.id}`}
            type="text"
            value={parcel.description}
            onChange={(e) => handleDescriptionChange(parcel.id, e.target.value)}
            autoComplete="off"
          />

          <label htmlFor={`parcel-quantity-${parcel.id}`}>
            {t.fieldParcelQuantity}
          </label>
          <input
            id={`parcel-quantity-${parcel.id}`}
            type="number"
            min={1}
            value={parcel.quantity}
            onChange={(e) => handleQuantityChange(parcel.id, e.target.value)}
          />

          <button
            type="button"
            aria-label={`${t.removeParcel} ${index + 1}`}
            onClick={() => handleRemove(parcel.id)}
          >
            {t.removeParcel}
          </button>
        </div>
      ))}

      <button type="button" onClick={handleAdd}>
        {t.addParcel}
      </button>
    </fieldset>
  );
};

// Narrow exported type to what the API actually needs (no client-side id on new records)
export type ParcelPayload = Omit<ParcelRow, 'id'> & { id?: string };

export function toParcelPayloads(parcels: ParcelRow[]): ParcelPayload[] {
  return parcels.map(({ id, description, quantity }) => ({
    // Only include id if it looks like a server-assigned id (not a UUID we generated locally).
    // Convention: server ids are numeric strings or UUIDs that exist in the DB.
    // For new parcels (client-generated UUID never sent to server), we omit id.
    // Implementer must verify with actual backend contract from task-04.
    id: id.startsWith('__new__') ? undefined : id,
    description,
    quantity,
  }));
}

export default ParcelsSection;
```

If the backend ignores unknown ids, sending the client UUID is harmless.
