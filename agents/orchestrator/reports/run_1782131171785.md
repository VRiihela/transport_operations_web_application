import { z } from 'zod';

// ... existing schemas remain unchanged above this line ...

export const updateJobScheduleSchema = z
  .object({
    scheduledStart: z.string().datetime({ message: 'scheduledStart must be a valid ISO 8601 datetime string' }).nullish(),
    scheduledEnd: z.string().datetime({ message: 'scheduledEnd must be a valid ISO 8601 datetime string' }).nullish(),
    schedulingNote: z
      .string()
      .max(500, { message: 'schedulingNote must not exceed 500 characters' })
      .nullish(),
  })
  // Refinement ①: at least one field must be present in the payload
  .refine(
    (data) => {
      const keys = Object.keys(data) as Array<keyof typeof data>;
      return keys.some((k) => data[k] !== undefined);
    },
    {
      message: 'At least one field (scheduledStart, scheduledEnd, schedulingNote) must be provided',
    }
  )
  // Refinement ②: if both scheduledStart and scheduledEnd resolve to null/undefined,
  // schedulingNote must be a non-empty string
  .refine(
    (data) => {
      const startCleared = data.scheduledStart == null;
      const endCleared = data.scheduledEnd == null;
      if (startCleared && endCleared) {
        return typeof data.schedulingNote === 'string' && data.schedulingNote.trim().length > 0;
      }
      return true;
    },
    {
      message:
        'schedulingNote must be a non-empty string when both scheduledStart and scheduledEnd are cleared to null',
      path: ['schedulingNote'],
    }
  )
  // Refinement ③: if both dates are non-null, scheduledEnd must be after scheduledStart
  .refine(
    (data) => {
      if (data.scheduledStart != null && data.scheduledEnd != null) {
        return new Date(data.scheduledEnd) > new Date(data.scheduledStart);
      }
      return true;
    },
    {
      message: 'scheduledEnd must be after scheduledStart',
      path: ['scheduledEnd'],
    }
  );

export type UpdateJobScheduleInput = z.infer<typeof updateJobScheduleSchema>;
```

**Important note on Refinement ①:** `.nullish()` fields that are absent from the payload will be `undefined` in the parsed object. The refinement checks that at least one key has a value that is not `undefined` — this correctly rejects `{}` while accepting `{ schedulingNote: null }` (a valid "clear the note" operation). This also means `{ scheduledStart: null, scheduledEnd: null }` without a note will pass Refinement ① but fail Refinement ②, which is the correct layered behaviour.

import { Request, Response } from 'express';
import { updateJobScheduleSchema } from '../schemas/job.schemas';
import { PrismaClientKnownRequestError } from '@prisma/client/runtime/library';
import * as jobService from '../services/job.service';

// ... existing handlers remain unchanged above this line ...

export async function updateJobSchedule(req: Request, res: Response): Promise<void> {
  // ID comes exclusively from the route parameter — never from the request body
  const { id } = req.params;

  // Validate and parse the request body
  const parseResult = updateJobScheduleSchema.safeParse(req.body);
  if (!parseResult.success) {
    res.status(400).json({ error: parseResult.error.errors[0]?.message ?? 'Invalid request body' });
    return;
  }

  const { scheduledStart, scheduledEnd, schedulingNote } = parseResult.data;

  // Build the update payload — only include fields explicitly provided in the request.
  // .nullish() means undefined = "not provided" (omit from update), null = "clear the field".
  const updateData: {
    scheduledStart?: string | null;
    scheduledEnd?: string | null;
    schedulingNote?: string | null;
  } = {};

  if (scheduledStart !== undefined) updateData.scheduledStart = scheduledStart ?? null;
  if (scheduledEnd !== undefined) updateData.scheduledEnd = scheduledEnd ?? null;
  if (schedulingNote !== undefined) updateData.schedulingNote = schedulingNote ?? null;

  try {
    const job = await jobService.updateJob(id, updateData);

    // Handle null return if jobService.updateJob returns null for not-found
    if (job === null) {
      res.status(404).json({ error: 'Job not found' });
      return;
    }

    res.status(200).json({ data: job });
  } catch (error) {
    // Handle Prisma record-not-found error (P2025)
    if (error instanceof PrismaClientKnownRequestError && error.code === 'P2025') {
      res.status(404).json({ error: 'Job not found' });
      return;
    }
    // Re-throw unexpected errors to the global error handler
    throw error;
  }
}
```

**Note on the update payload shape:** `jobService.updateJob` accepts partial fields. By only including keys where the value is not `undefined`, we avoid accidentally overwriting DB fields that were not part of this request. The `?? null` coercion collapses Zod's `undefined | null` union to `null` for the DB layer, which expects `null` for a clear operation.

## Review of PATCH /api/jobs/:id/schedule Implementation

### 1. Correctness

**Refinement ① — "at least one field must be present" — is BROKEN**

The `.refine()` checks `Object.keys(data).some(k => data[k] !== undefined)`. However, Zod's `.object()` by default strips unknown keys and includes all declared keys in the parsed output — even absent ones — as `undefined`. The check `Object.keys(data)` will return `['scheduledStart', 'scheduledEnd', 'schedulingNote']` regardless of what was sent, and all three will be `undefined` for an empty body `{}`. So the refinement result would be `false` correctly — but let me reason more carefully: with `z.object({...})` and no `.strict()`, Zod strips extra keys but does NOT add missing keys. Absent `.nullish()` keys remain absent from the output object. So `Object.keys({})` → empty array → `.some()` → `false` → refinement fires. **This actually works.** However, the logic is subtle enough to be risky. A more idiomatic check would be:

```ts
(data) => data.scheduledStart !== undefined || data.scheduledEnd !== undefined || data.schedulingNote !== undefined
```

This is clearer and avoids the `keyof typeof data` cast.

**Refinement ② — whitespace-only `schedulingNote` (`'   '`)**

The refinement checks `data.schedulingNote.trim().length > 0`. TC-04b tests this with `'   '`. The Zod schema for `schedulingNote` is `z.string().max(500).nullish()` — there is **no `.trim()` applied to the stored value**, only to the check. This means `'   '` would fail validation (correct) but if someone sends `' a '`, it would pass and the whitespace would be stored in the DB.
