// frontend/src/pages/DispatcherBoard/formatSchedule.ts

export interface ScheduleFields {
  scheduledStart: string | null;
  scheduledEnd: string | null;
  scheduleType: string | null;
  schedulingNote: string | null;
}

export interface FormatScheduleResult {
  primary: string;
  label: string;
}

/**
 * Formats time components of a date string to "HH:MM" using local time.
 * Pure function — no side effects, no external dependencies.
 */
function toHHMM(isoString: string): string {
  const date = new Date(isoString);
  const hours = String(date.getHours()).padStart(2, "0");
  const minutes = String(date.getMinutes()).padStart(2, "0");
  return `${hours}:${minutes}`;
}

/**
 * Computes the whole number of hours between two ISO datetime strings.
 * Uses Math.round for sub-hour rounding (e.g. 90 min → 2 h, 89 min → 1 h).
 */
function diffWholeHours(startIso: string, endIso: string): number {
  const startMs = new Date(startIso).getTime();
  const endMs = new Date(endIso).getTime();
  const diffMs = endMs - startMs;
  return Math.round(diffMs / (1000 * 60 * 60));
}

/**
 * Pure formatting helper — no React or Axios imports.
 *
 * Priority order:
 *   1. Both null           → TBC
 *   2. scheduledEnd null   → fixed single time (regardless of scheduleType)
 *   3. scheduleType WINDOW → arrival window range with en dash
 *   4. scheduleType DURATION → duration range with arrow and computed hours
 *   5. Fallback            → treat as fixed single time
 */
export function formatSchedule(job: ScheduleFields): FormatScheduleResult {
  const { scheduledStart, scheduledEnd, scheduleType, schedulingNote } = job;

  // Branch 1: both times absent → TBC
  if (scheduledStart === null && scheduledEnd === null) {
    return {
      primary: schedulingNote ?? "TBC",
      label: "TBC",
    };
  }

  // Branch 2: end absent → fixed time (legacy rows, or genuinely fixed)
  if (scheduledEnd === null) {
    // scheduledStart is non-null here (branch 1 already handled both-null)
    return {
      primary: toHHMM(scheduledStart as string),
      label: "kiinteä aika",
    };
  }

  // From here both scheduledStart and scheduledEnd are non-null strings.
  // scheduledStart could theoretically be null if only end is set — treat as fixed on end.
  const start = scheduledStart ?? scheduledEnd;

  // Branch 3: arrival window
  if (scheduleType === "WINDOW") {
    return {
      primary: `${toHHMM(start)}–${toHHMM(scheduledEnd)}`,
      label: "saapumisaika",
    };
  }

  // Branch 4: duration
  if (scheduleType === "DURATION") {
    const hours = diffWholeHours(start, scheduledEnd);
    return {
      primary: `${toHHMM(start)} → ${toHHMM(scheduledEnd)}`,
      label: `kesto · ${hours} h`,
    };
  }

  // Branch 5: defensive fallback — treat as fixed using start (or end if start absent)
  return {
    primary: toHHMM(start),
    label: "kiinteä aika",
  };
}
